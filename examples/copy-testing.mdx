## Overview

This example demonstrates how to use Semilattice to optimize blog post titles by testing different variations with your target audience. The script generates multiple title options, tests them in batches, and iteratively refines the selection to find the most appealing copy.

## The Process

1. **Generate initial title ideas** - Create a diverse set of potential blog post titles
2. **Test in batches** - Run groups of 3 titles through Semilattice's simulation
3. **Select winners** - Choose the best performing title from each batch
4. **Create shortlist** - Compile the winners into a final shortlist
5. **Final testing** - Test the shortlist to determine the ultimate winner

## Complete Example Script

```python
import os
import time
import random
from semilattice import Semilattice

# Initialize the client
semilattice = Semilattice(
    api_key=os.environ.get("SEMILATTICE_API_KEY"),
)

# Your target population ID (replace with your actual population)
POPULATION_ID = "your-population-uuid-here"

def fake_llm_generate_titles(topic, count=9):
    """
    Simulate an LLM generating blog post titles.
    In practice, you'd call your actual LLM here.
    """
    base_titles = [
        f"The Ultimate Guide to {topic}",
        f"How to Master {topic} in 30 Days",
        f"Why {topic} is Changing Everything",
        f"The Secret to {topic} Success",
        f"Common {topic} Mistakes to Avoid",
        f"The Future of {topic} is Here",
        f"Unlock the Power of {topic}",
        f"From Zero to {topic} Hero",
        f"The {topic} Revolution Begins",
        f"Transform Your Business with {topic}",
        f"The Complete {topic} Playbook",
        f"Why Everyone is Talking About {topic}",
    ]
    
    return random.sample(base_titles, min(count, len(base_titles)))

def test_titles_batch(titles, question_context="Which blog post title would you be most likely to click on?"):
    """
    Test a batch of titles using Semilattice's simulation.
    """
    print(f"Testing batch: {titles}")
    
    result = semilattice.answers.simulate(
        population_id=POPULATION_ID,
        answers={
            "question": question_context,
            "question_options": {"question_type": "single-choice"},
            "answer_options": titles
        }
    )
    
    answer_id = result.data[0].id
    
    # Poll for results
    while True:
        answer = semilattice.answers.get(answer_id)
        
        if answer.data.status == 'Predicted':
            print("✓ Simulation complete!")
            percentages = answer.data.simulated_answer_percentages
            
            # Find the winner
            winner = max(percentages.items(), key=lambda x: x[1])
            print(f"Winner: '{winner[0]}' with {winner[1]:.1%} preference")
            print(f"Full results: {percentages}")
            print()
            
            return winner[0], percentages
            
        elif answer.data.status == 'Failed':
            print("❌ Simulation failed")
            return None, None
            
        print(f"Status: {answer.data.status}")
        time.sleep(5)

def optimize_blog_titles(topic):
    """
    Complete workflow to optimize blog post titles.
    """
    print(f"🎯 Optimizing blog post titles for: {topic}")
    print("=" * 50)
    
    # Step 1: Generate initial titles
    print("📝 Step 1: Generating initial title ideas...")
    all_titles = fake_llm_generate_titles(topic, 9)
    print(f"Generated {len(all_titles)} title options")
    print()
    
    # Step 2: Test in batches of 3
    print("🧪 Step 2: Testing titles in batches of 3...")
    winners = []
    
    for i in range(0, len(all_titles), 3):
        batch = all_titles[i:i+3]
        if len(batch) < 3 and len(all_titles) > 3:
            # If we have a small remainder, add it to the previous batch
            continue
            
        winner, percentages = test_titles_batch(batch)
        if winner:
            winners.append(winner)
    
    print(f"🏆 Batch winners: {len(winners)} titles selected")
    for i, winner in enumerate(winners, 1):
        print(f"  {i}. {winner}")
    print()
    
    # Step 3: Final showdown
    if len(winners) > 1:
        print("🥊 Step 3: Final showdown between winners...")
        final_winner, final_percentages = test_titles_batch(
            winners, 
            "Which blog post title would be most compelling for your audience?"
        )
        
        print("🎉 FINAL RESULT:")
        print(f"Best title: '{final_winner}'")
        print(f"Audience preference: {final_percentages[final_winner]:.1%}")
        
        return final_winner, final_percentages
    else:
        print("🎉 FINAL RESULT:")
        print(f"Best title: '{winners[0]}'")
        return winners[0], None

# Example usage
if __name__ == "__main__":
    # Test with different topics
    topics = ["AI Marketing", "Remote Work", "Sustainable Business"]
    
    for topic in topics:
        try:
            best_title, results = optimize_blog_titles(topic)
            print(f"✅ Optimization complete for '{topic}'")
            print(f"Recommended title: {best_title}")
            print("\n" + "="*70 + "\n")
            
        except Exception as e:
            print(f"❌ Error optimizing titles for '{topic}': {e}")
            print("\n" + "="*70 + "\n")
```

## Key Insights

This example demonstrates several powerful patterns:

1. **Iterative Testing** - Testing in small batches allows for more focused comparisons
2. **Population Targeting** - Different audiences may prefer different styles of titles
3. **Multi-Criteria Evaluation** - Test for different factors (appeal, trust, excitement)
4. **Continuous Optimization** - Use results to generate better variations
